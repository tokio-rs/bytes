21/// A unique reference to a contiguous slice of memory.
22///
23/// `BytesMut` represents a unique view into a potentially shared memory region.
24/// Given the uniqueness guarantee, owners of `BytesMut` handles are able to
25/// mutate the memory.
26///
27/// `BytesMut` can be thought of as containing a `buf: Arc<Vec<u8>>`, an offset
28/// into `buf`, a slice length, and a guarantee that no other `BytesMut` for the
29/// same `buf` overlaps with its slice. That guarantee means that a write lock
30/// is not required.
31///
32/// # Growth
33///
34/// `BytesMut`'s `BufMut` implementation will implicitly grow its buffer as
35/// necessary. However, explicitly reserving the required space up-front before
36/// a series of inserts will be more efficient.
37///
38/// # Examples
39///
40/// ```
41/// use bytes::{BytesMut, BufMut};
42///
43/// let mut buf = BytesMut::with_capacity(64);
44///
45/// buf.put_u8(b'h');
46/// buf.put_u8(b'e');
47/// buf.put(&b"llo"[..]);
48///
49/// assert_eq!(&buf[..], b"hello");
50///
51/// // Freeze the buffer so that it can be shared
52/// let a = buf.freeze();
53///
54/// // This does not allocate, instead `b` points to the same memory.
55/// let b = a.clone();
56///
57/// assert_eq!(&a[..], b"hello");
58/// assert_eq!(&b[..], b"hello");
59/// ```
60pub struct BytesMut {
61    ptr: NonNull<u8>,
62    len: usize,
63    cap: usize,
64    data: *mut Shared,
65}

67// Thread-safe reference-counted container for the shared storage. This mostly
68// the same as `core::sync::Arc` but without the weak counter. The ref counting
69// fns are based on the ones found in `std`.
70//
71// The main reason to use `Shared` instead of `core::sync::Arc` is that it ends
72// up making the overall code simpler and easier to reason about. This is due to
73// some of the logic around setting `Inner::arc` and other ways the `arc` field
74// is used. Using `Arc` ended up requiring a number of funky transmutes and
75// other shenanigans to make it work.
76struct Shared {
77    vec: Vec<u8>,
78    original_capacity_repr: usize,
79    ref_count: AtomicUsize,
80}

235    ///
236    /// The conversion is zero cost and is used to indicate that the slice

259    pub fn freeze(self) -> Bytes {
260        let bytes = ManuallyDrop::new(self);
261        if bytes.kind() == KIND_VEC {
262            // Just re-use `Bytes` internal Vec vtable
263            unsafe {
264                let off = bytes.get_vec_pos();
265                let vec = rebuild_vec(bytes.ptr.as_ptr(), bytes.len, bytes.cap, off);
266                let mut b: Bytes = vec.into();
267                b.advance(off);
268                b
269            }
270        } else {
271            debug_assert_eq!(bytes.kind(), KIND_ARC);
272
273            let ptr = bytes.ptr.as_ptr();
274            let len = bytes.len;
275            let data = AtomicPtr::new(bytes.data.cast());
276            unsafe { Bytes::with_vtable(ptr, len, data, &SHARED_VTABLE) }
277        }
278    }

304    /// Splits the bytes into two at the given index.
305    ///
306    /// Afterwards `self` contains elements `[0, at)`, and the returned
307    /// `BytesMut` contains elements `[at, capacity)`. It's guaranteed that the
308    /// memory does not move, that is, the address of `self` does not change,
309    /// and the address of the returned slice is `at` bytes after that.
310    ///
311    /// This is an `O(1)` operation that just increases the reference count
312    /// and sets a few indices.
313    ///
314    /// # Examples
315    ///
316    /// ```
317    /// use bytes::BytesMut;
318    ///
319    /// let mut a = BytesMut::from(&b"hello world"[..]);
320    /// let mut b = a.split_off(5);
321    ///
322    /// a[0] = b'j';
323    /// b[0] = b'!';
324    ///
325    /// assert_eq!(&a[..], b"jello");
326    /// assert_eq!(&b[..], b"!world");
327    /// ```
328    ///
329    /// # Panics
330    ///
331    /// Panics if `at > capacity`.
332    #[must_use = "consider BytesMut::truncate if you don't need the other half"]
333    pub fn split_off(&mut self, at: usize) -> BytesMut {
334        assert!(
335            at <= self.capacity(),
336            "split_off out of bounds: {:?} <= {:?}",
337            at,
338            self.capacity(),
339        );
340        unsafe {
341            // SAFETY: `shallow_clone` increments the reference count (or
342            // promotes to shared) and returns a bitwise copy of the handle.
343            // The caller immediately adjusts both handles so they represent
344            // disjoint regions.
345            let mut other = self.shallow_clone();
346            // SAFETY: We've checked that `at` <= `self.capacity()` above.
347            other.advance_unchecked(at);
348            self.cap = at;
349            self.len = cmp::min(self.len, at);
350            other
351        }
352    }

478    /// Resizes the buffer so that `len` is equal to `new_len`.
479    ///
480    /// If `new_len` is greater than `len`, the buffer is extended by the
481    /// difference with each additional byte set to `value`. If `new_len` is
482    /// less than `len`, the buffer is simply truncated.
483    ///
484    /// # Examples
485    ///
486    /// ```
487    /// use bytes::BytesMut;
488    ///
489    /// let mut buf = BytesMut::new();
490    ///
491    /// buf.resize(3, 0x1);
492    /// assert_eq!(&buf[..], &[0x1, 0x1, 0x1]);
493    ///
494    /// buf.resize(2, 0x2);
495    /// assert_eq!(&buf[..], &[0x1, 0x1]);
496    ///
497    /// buf.resize(4, 0x3);
498    /// assert_eq!(&buf[..], &[0x1, 0x1, 0x3, 0x3]);
499    /// ```
500    pub fn resize(&mut self, new_len: usize, value: u8) {
501        let additional = if let Some(additional) = new_len.checked_sub(self.len()) {
502            additional
503        } else {
504            self.truncate(new_len);
505            return;
506        };
507
508        if additional == 0 {
509            return;
510        }
511
512        self.reserve(additional);
513        let dst = self.spare_capacity_mut().as_mut_ptr();
514        // SAFETY: `spare_capacity_mut` returns a valid, properly aligned pointer and we've
515        // reserved enough space to write `additional` bytes.
516        unsafe { ptr::write_bytes(dst, value, additional) };
517
518        // SAFETY: There are at least `new_len` initialized bytes in the buffer so no
519        // uninitialized bytes are being exposed.
520        unsafe { self.set_len(new_len) };
521    }

554    /// Reserves capacity for at least `additional` more bytes to be inserted
555    /// into the given `BytesMut`.
556    ///
557    /// More than `additional` bytes may be reserved in order to avoid frequent
558    /// reallocations. A call to `reserve` may result in an allocation.
559    ///
560    /// Before allocating new buffer space, the function will attempt to reclaim
561    /// space in the existing buffer. If the current handle references a view
562    /// into a larger original buffer, and all other handles referencing part
563    /// of the same original buffer have been dropped, then the current view
564    /// can be copied/shifted to the front of the buffer and the handle can take
565    /// ownership of the full buffer, provided that the full buffer is large
566    /// enough to fit the requested additional capacity.
567    ///
568    /// This optimization will only happen if shifting the data from the current
569    /// view to the front of the buffer is not too expensive in terms of the
570    /// (amortized) time required. The precise condition is subject to change;
571    /// as of now, the length of the data being shifted needs to be at least as
572    /// large as the distance that it's shifted by. If the current view is empty
573    /// and the original buffer is large enough to fit the requested additional
574    /// capacity, then reallocations will never happen.
575    ///
576    /// This method does not preserve data stored in the unused capacity.
577    ///
578    /// # Examples
579    ///
580    /// In the following example, a new buffer is allocated.
581    ///
582    /// ```
583    /// use bytes::BytesMut;
584    ///
585    /// let mut buf = BytesMut::from(&b"hello"[..]);
586    /// buf.reserve(64);
587    /// assert!(buf.capacity() >= 69);
588    /// ```
589    ///
590    /// In the following example, the existing buffer is reclaimed.
591    ///
592    /// ```
593    /// use bytes::{BytesMut, BufMut};
594    ///
595    /// let mut buf = BytesMut::with_capacity(128);
596    /// buf.put(&[0; 64][..]);
597    ///
598    /// let ptr = buf.as_ptr();
599    /// let other = buf.split();
600    ///
601    /// assert!(buf.is_empty());
602    /// assert_eq!(buf.capacity(), 64);
603    ///
604    /// drop(other);
605    /// buf.reserve(128);
606    ///
607    /// assert_eq!(buf.capacity(), 128);
608    /// assert_eq!(buf.as_ptr(), ptr);
609    /// ```
610    ///
611    /// # Panics
612    ///
613    /// Panics if the new capacity overflows `usize`.
614    #[inline]
615    pub fn reserve(&mut self, additional: usize) {
616        let len = self.len();
617        let rem = self.capacity() - len;
618
619        if additional <= rem {
620            // The handle can already store at least `additional` more bytes, so
621            // there is no further work needed to be done.
622            return;
623        }
624
625        // will always succeed
626        let _ = self.reserve_inner(additional, true);
627    }

629    // In separate function to allow the short-circuits in `reserve` and `try_reclaim` to
630    // be inline-able. Significantly helps performance. Returns false if it did not succeed.
631    fn reserve_inner(&mut self, additional: usize, allocate: bool) -> bool {
632        let len = self.len();
633        let kind = self.kind();
634
635        if kind == KIND_VEC {
636            // If there's enough free space before the start of the buffer, then
637            // just copy the data backwards and reuse the already-allocated
638            // space.
639            //
640            // Otherwise, since backed by a vector, use `Vec::reserve`
641            //
642            // We need to make sure that this optimization does not kill the
643            // amortized runtimes of BytesMut's operations.
644            unsafe {
645                let off = self.get_vec_pos();
646
647                // Only reuse space if we can satisfy the requested additional space.
648                //
649                // Also check if the value of `off` suggests that enough bytes
650                // have been read to account for the overhead of shifting all
651                // the data (in an amortized analysis).
652                // Hence the condition `off >= self.len()`.
653                //
654                // This condition also already implies that the buffer is going
655                // to be (at least) half-empty in the end; so we do not break
656                // the (amortized) runtime with future resizes of the underlying
657                // `Vec`.
658                //
659                // [For more details check issue #524, and PR #525.]
660                if self.capacity() - self.len() + off >= additional && off >= self.len() {
661                    // There's enough space, and it's not too much overhead:
662                    // reuse the space!
663                    //
664                    // Just move the pointer back to the start after copying
665                    // data back.
666                    let base_ptr = self.ptr.as_ptr().sub(off);
667                    // Since `off >= self.len()`, the two regions don't overlap.
668                    ptr::copy_nonoverlapping(self.ptr.as_ptr(), base_ptr, self.len);
669                    self.ptr = vptr(base_ptr);
670                    self.set_vec_pos(0);
671
672                    // Length stays constant, but since we moved backwards we
673                    // can gain capacity back.
674                    self.cap += off;
675                } else {
676                    if !allocate {
677                        return false;
678                    }
679                    // Not enough space, or reusing might be too much overhead:
680                    // allocate more space!
681                    let mut v =
682                        ManuallyDrop::new(rebuild_vec(self.ptr.as_ptr(), self.len, self.cap, off));
683                    v.reserve(additional);
684
685                    // Update the info
686                    self.ptr = vptr(v.as_mut_ptr().add(off));
687                    self.cap = v.capacity() - off;
688                    debug_assert_eq!(self.len, v.len() - off);
689                }
690
691                return true;
692            }
693        }
694
695        debug_assert_eq!(kind, KIND_ARC);
696        let shared: *mut Shared = self.data;
697
698        // Reserving involves abandoning the currently shared buffer and
699        // allocating a new vector with the requested capacity.
700        //
701        // Compute the new capacity
702        let mut new_cap = match len.checked_add(additional) {
703            Some(new_cap) => new_cap,
704            None if !allocate => return false,
705            None => panic!("overflow"),
706        };
707
708        unsafe {
709            // First, try to reclaim the buffer. This is possible if the current
710            // handle is the only outstanding handle pointing to the buffer.
711            if (*shared).is_unique() {
712                // This is the only handle to the buffer. It can be reclaimed.
713                // However, before doing the work of copying data, check to make
714                // sure that the vector has enough capacity.
715                let v = &mut (*shared).vec;
716
717                let v_capacity = v.capacity();
718                let ptr = v.as_mut_ptr();
719
720                let offset = self.ptr.as_ptr().offset_from(ptr) as usize;
721
722                let new_cap_plus_offset = match new_cap.checked_add(offset) {
723                    Some(new_cap_plus_offset) => new_cap_plus_offset,
724                    None if !allocate => return false,
725                    None => panic!("overflow"),
726                };
727
728                // Compare the condition in the `kind == KIND_VEC` case above
729                // for more details.
730                if v_capacity >= new_cap_plus_offset {
731                    self.cap = new_cap;
732                    // no copy is necessary
733                } else if v_capacity >= new_cap && offset >= len {
734                    // The capacity is sufficient, and copying is not too much
735                    // overhead: reclaim the buffer!
736
737                    // `offset >= len` means: no overlap
738                    ptr::copy_nonoverlapping(self.ptr.as_ptr(), ptr, len);
739
740                    self.ptr = vptr(ptr);
741                    self.cap = v.capacity();
742                } else {
743                    if !allocate {
744                        return false;
745                    }
746
747                    // new_cap is calculated in terms of `BytesMut`, not the underlying
748                    // `Vec`, so it does not take the offset into account.
749                    //
750                    // Thus we have to manually add it here.
751                    new_cap = new_cap_plus_offset;
752
753                    // The vector capacity is not sufficient. The reserve request is
754                    // asking for more than the initial buffer capacity. Allocate more
755                    // than requested if `new_cap` is not much bigger than the current
756                    // capacity.
757                    //
758                    // There are some situations, using `reserve_exact` that the
759                    // buffer capacity could be below `original_capacity`, so do a
760                    // check.
761                    let double = v.capacity().checked_shl(1).unwrap_or(new_cap);
762
763                    new_cap = cmp::max(double, new_cap);
764
765                    // No space - allocate more
766                    //
767                    // The length field of `Shared::vec` is not used by the `BytesMut`;
768                    // instead we use the `len` field in the `BytesMut` itself. However,
769                    // when calling `reserve`, it doesn't guarantee that data stored in
770                    // the unused capacity of the vector is copied over to the new
771                    // allocation, so we need to ensure that we don't have any data we
772                    // care about in the unused capacity before calling `reserve`.
773                    debug_assert!(offset + len <= v.capacity());
774                    v.set_len(offset + len);
775                    v.reserve(new_cap - v.len());
776
777                    // Update the info
778                    self.ptr = vptr(v.as_mut_ptr().add(offset));
779                    self.cap = v.capacity() - offset;
780                }
781
782                return true;
783            }
784        }
785        if !allocate {
786            return false;
787        }
788
789        let original_capacity_repr = unsafe { (*shared).original_capacity_repr };
790        let original_capacity = original_capacity_from_repr(original_capacity_repr);
791
792        new_cap = cmp::max(new_cap, original_capacity);
793
794        // Create a new vector to store the data
795        let mut v = ManuallyDrop::new(Vec::with_capacity(new_cap));
796
797        // Copy the bytes
798        v.extend_from_slice(self.as_ref());
799
800        // Release the shared handle. This must be done *after* the bytes are
801        // copied.
802        unsafe { release_shared(shared) };
803
804        // Update self
805        let data = (original_capacity_repr << ORIGINAL_CAPACITY_OFFSET) | KIND_VEC;
806        self.data = invalid_ptr(data);
807        self.ptr = vptr(v.as_mut_ptr());
808        self.cap = v.capacity();
809        debug_assert_eq!(self.len, v.len());
810        true
811    }

813    /// Attempts to cheaply reclaim already allocated capacity for at least `additional` more
814    /// bytes to be inserted into the given `BytesMut` and returns `true` if it succeeded.
815    ///
816    /// `try_reclaim` behaves exactly like `reserve`, except that it never allocates new storage
817    /// and returns a `bool` indicating whether it was successful in doing so:
818    ///
819    /// `try_reclaim` returns false under these conditions:
820    ///  - The spare capacity left is less than `additional` bytes AND
821    ///  - The existing allocation cannot be reclaimed cheaply or it was less than
822    ///    `additional` bytes in size
823    ///
824    /// Reclaiming the allocation cheaply is possible if the `BytesMut` has no outstanding
825    /// references through other `BytesMut`s or `Bytes` which point to the same underlying
826    /// storage.
827    ///
828    /// This method does not preserve data stored in the unused capacity.
829    ///
830    /// # Examples
831    ///
832    /// ```
833    /// use bytes::BytesMut;
834    ///
835    /// let mut buf = BytesMut::with_capacity(64);
836    /// assert_eq!(true, buf.try_reclaim(64));
837    /// assert_eq!(64, buf.capacity());
838    ///
839    /// buf.extend_from_slice(b"abcd");
840    /// let mut split = buf.split();
841    /// assert_eq!(60, buf.capacity());
842    /// assert_eq!(4, split.capacity());
843    /// assert_eq!(false, split.try_reclaim(64));
844    /// assert_eq!(false, buf.try_reclaim(64));
845    /// // The split buffer is filled with "abcd"
846    /// assert_eq!(false, split.try_reclaim(4));
847    /// // buf is empty and has capacity for 60 bytes
848    /// assert_eq!(true, buf.try_reclaim(60));
849    ///
850    /// drop(buf);
851    /// assert_eq!(false, split.try_reclaim(64));
852    ///
853    /// split.clear();
854    /// assert_eq!(4, split.capacity());
855    /// assert_eq!(true, split.try_reclaim(64));
856    /// assert_eq!(64, split.capacity());
857    /// ```
858    // I tried splitting out try_reclaim_inner after the short circuits, but it was inlined
859    // regardless with Rust 1.78.0 so probably not worth it
860    #[inline]
861    #[must_use = "consider BytesMut::reserve if you need an infallible reservation"]
862    pub fn try_reclaim(&mut self, additional: usize) -> bool {
863        let len = self.len();
864        let rem = self.capacity() - len;
865
866        if additional <= rem {
867            // The handle can already store at least `additional` more bytes, so
868            // there is no further work needed to be done.
869            return true;
870        }
871
872        self.reserve_inner(additional, false)
873    }

929
930        let cnt = end - begin;

932

935        let dst = self.spare_capacity_mut();
936

938        unsafe { ptr::copy_nonoverlapping(src, dst.as_mut_ptr().cast(), cnt) }
939
940        // SAFETY: capacity is already reserved and filled with data
941        unsafe { self.advance_mut(cnt) }

1016    /// Advance the buffer without bounds checking.
1017    ///
1018    /// # SAFETY
1019    ///
1020    /// The caller must ensure that `count` <= `self.cap`.
1021    pub(crate) unsafe fn advance_unchecked(&mut self, count: usize) {
1022        // Setting the start to 0 is a no-op, so return early if this is the
1023        // case.
1024        if count == 0 {
1025            return;
1026        }
1027
1028        debug_assert!(count <= self.cap, "internal: set_start out of bounds");
1029
1030        let kind = self.kind();
1031
1032        if kind == KIND_VEC {
1033            // Setting the start when in vec representation is a little more
1034            // complicated. First, we have to track how far ahead the
1035            // "start" of the byte buffer from the beginning of the vec. We
1036            // also have to ensure that we don't exceed the maximum shift.
1037            let pos = self.get_vec_pos() + count;
1038
1039            if pos <= MAX_VEC_POS {
1040                self.set_vec_pos(pos);
1041            } else {
1042                // The repr must be upgraded to ARC. This will never happen
1043                // on 64 bit systems and will only happen on 32 bit systems
1044                // when shifting past 134,217,727 bytes. As such, we don't
1045                // worry too much about performance here.
1046                self.promote_to_shared(/*ref_count = */ 1);
1047            }
1048        }
1049
1050        // Updating the start of the view is setting `ptr` to point to the
1051        // new start and updating the `len` field to reflect the new length
1052        // of the view.
1053        self.ptr = vptr(self.ptr.as_ptr().add(count));
1054        self.len = self.len.saturating_sub(count);
1055        self.cap -= count;
1056    }

1058    /// Absorbs a `BytesMut` that was previously split off.
1059    ///
1060    /// If the two `BytesMut` objects were previously contiguous, i.e., if
1061    /// `other` was created by calling `split_off` on this `BytesMut`, then
1062    /// this is an `O(1)` operation that just decreases a reference
1063    /// count and sets a few indices. Otherwise this method returns an error
1064    /// containing the original `other`.
1065    ///
1066    /// # Examples
1067    ///
1068    /// ```
1069    /// use bytes::BytesMut;
1070    ///
1071    /// let mut buf = BytesMut::with_capacity(64);
1072    /// buf.extend_from_slice(b"aaabbbcccddd");
1073    ///
1074    /// let mut split_1 = buf.split_off(3);
1075    /// let split_2 = split_1.split_off(3);
1076    /// assert_eq!(b"aaa", &buf[..]);
1077    /// assert_eq!(b"bbb", &split_1[..]);
1078    /// assert_eq!(b"cccddd", &split_2[..]);
1079    ///
1080    /// let split_2 = buf.try_unsplit(split_2).unwrap_err();
1081    ///
1082    /// buf.try_unsplit(split_1).unwrap();
1083    /// buf.try_unsplit(split_2).unwrap();
1084    /// assert_eq!(b"aaabbbcccddd", &buf[..]);
1085    /// ```
1086    pub fn try_unsplit(&mut self, other: BytesMut) -> Result<(), BytesMut> {
1087        if other.capacity() == 0 {
1088            return Ok(());
1089        }
1090
1091        let ptr = unsafe { self.ptr.as_ptr().add(self.len) };
1092        if ptr == other.ptr.as_ptr()
1093            && self.kind() == KIND_ARC
1094            && other.kind() == KIND_ARC
1095            && self.data == other.data
1096        {
1097            // Contiguous blocks, just combine directly
1098            self.len += other.len;
1099            self.cap += other.cap;
1100            Ok(())
1101        } else {
1102            Err(other)
1103        }
1104    }

1111    unsafe fn promote_to_shared(&mut self, ref_cnt: usize) {
1112        debug_assert_eq!(self.kind(), KIND_VEC);
1113        debug_assert!(ref_cnt == 1 || ref_cnt == 2);
1114
1115        let original_capacity_repr =
1116            (self.data as usize & ORIGINAL_CAPACITY_MASK) >> ORIGINAL_CAPACITY_OFFSET;
1117
1118        // The vec offset cannot be concurrently mutated, so there
1119        // should be no danger reading it.
1120        let off = (self.data as usize) >> VEC_POS_OFFSET;
1121
1122        // First, allocate a new `Shared` instance containing the
1123        // `Vec` fields. It's important to note that `ptr`, `len`,
1124        // and `cap` cannot be mutated without having `&mut self`.
1125        // This means that these fields will not be concurrently
1126        // updated and since the buffer hasn't been promoted to an
1127        // `Arc`, those three fields still are the components of the
1128        // vector.
1129        //
1130        // Explicitly allocate before invoking rebuild_vec() so that
1131        // the vector is not dropped if Box::new() panics.
1132        let shared = Box::new(MaybeUninit::<Shared>::uninit());
1133        let shared = Shared::init_to_raw(
1134            shared,
1135            Shared {
1136                vec: rebuild_vec(self.ptr.as_ptr(), self.len, self.cap, off),
1137                original_capacity_repr,
1138                ref_count: AtomicUsize::new(ref_cnt),
1139            },
1140        );
1141
1142        // The pointer should be aligned, so this assert should
1143        // always succeed.
1144        debug_assert_eq!(shared as usize & KIND_MASK, KIND_ARC);
1145
1146        self.data = shared;
1147    }

1222    /// Goes through `Vec::extend`, which carries std's `TrustedLen`
1223    /// specialization (slice-like iterators become a memcpy).
1224    ///

1226    ///
1227    /// The caller must ensure that `self.kind()` is `KIND_VEC`.

1230        debug_assert_eq!(self.kind(), KIND_VEC);
1231
1232        let off = self.get_vec_pos();
1233        let spare = self.cap - self.len;
1234        let mut vec = ManuallyDrop::new(rebuild_vec(self.ptr.as_ptr(), self.len, self.cap, off));

1238        self.len == self.cap
1239    }

1334                drop(src);
1335                match src_copy.try_into_mut() {
1336                    Ok(bytes_mut) => *dst = bytes_mut,
1337                    Err(bytes) => dst.extend_from_slice(&bytes),
1338                }
1339            }
1340
1341            put_reuse_allocation(self, src);
1342        } else {
1343            // In case the src isn't contiguous, reserve upfront.
1344            if src.chunk().len() < src.remaining() {
1345                self.reserve(src.remaining());
1346            }
1347
1348            while src.has_remaining() {
1349                let s = src.chunk();
1350                let l = s.len();
1351                self.extend_from_slice(s);
1352                src.advance(l);
1353            }
1354        }
1355    }
1356
1357    #[inline]
1358    fn put_slice(&mut self, src: &[u8]) {
1359        self.extend_from_slice(src);
1360    }
1361
1362    #[inline]
1363    fn put_u8(&mut self, n: u8) {
1364        // Same shape as `Vec::push`, avoiding the default's 1-byte `put_slice`.
1365        self.reserve(1);
1366        let len = self.len();
1367        // SAFETY: `reserve(1)` guarantees `len < capacity`, so `ptr + len` is
1368        // in bounds; the byte is initialized before `len` is bumped over it.
1369        unsafe {
1370            self.ptr.as_ptr().add(len).write(n);
1371            self.set_len(len + 1);
1372        }
1373    }
1374
1375    #[inline]
1376    fn put_i8(&mut self, n: i8) {
1377        self.put_u8(n as u8);
1378    }
1379
1380    #[inline]
1381    fn put_bytes(&mut self, val: u8, cnt: usize) {
1382        self.reserve(cnt);
1383        unsafe {
1384            let dst = self.spare_capacity_mut();
1385            // Reserved above
1386            debug_assert!(dst.len() >= cnt);
1387
1388            ptr::write_bytes(dst.as_mut_ptr(), val, cnt);
1389
1390            self.advance_mut(cnt);
1391        }
1392    }

1595unsafe fn increment_shared(ptr: *mut Shared) {
1596    let old_size = (*ptr).ref_count.fetch_add(1, Ordering::Relaxed);
1597
1598    if old_size > isize::MAX as usize {
1599        crate::abort();
1600    }
1601}
1602
1603unsafe fn release_shared(ptr: *mut Shared) {
1604    // `Shared` storage... follow the drop steps from Arc.
1605    if (*ptr).ref_count.fetch_sub(1, Ordering::Release) != 1 {
1606        return;
1607    }
1608
1609    // This fence is needed to prevent reordering of use of the data and
1610    // deletion of the data.  Because it is marked `Release`, the decreasing
1611    // of the reference count synchronizes with this `Acquire` fence. This
1612    // means that use of the data happens before decreasing the reference
1613    // count, which happens before this fence, which happens before the
1614    // deletion of the data.
1615    //
1616    // As explained in the [Boost documentation][1],
1617    //
1618    // > It is important to enforce any possible access to the object in one
1619    // > thread (through an existing reference) to *happen before* deleting
1620    // > the object in a different thread. This is achieved by a "release"
1621    // > operation after dropping a reference (any access to the object
1622    // > through this reference must obviously happened before), and an
1623    // > "acquire" operation before deleting the object.
1624    //
1625    // [1]: (www.boost.org/doc/libs/1_55_0/doc/html/atomic/usage_examples.html)
1626    //
1627    // Thread sanitizer does not support atomic fences. Use an atomic load
1628    // instead.
1629    (*ptr).ref_count.load(Ordering::Acquire);
1630
1631    // Drop the data
1632    drop(Box::from_raw(ptr));
1633}
1634
1635impl Shared {
1636    fn is_unique(&self) -> bool {
1637        // The goal is to check if the current handle is the only handle
1638        // that currently has access to the buffer. This is done by
1639        // checking if the `ref_count` is currently 1.
1640        //
1641        // The `Acquire` ordering synchronizes with the `Release` as
1642        // part of the `fetch_sub` in `release_shared`. The `fetch_sub`
1643        // operation guarantees that any mutations done in other threads
1644        // are ordered before the `ref_count` is decremented. As such,
1645        // this `Acquire` will guarantee that those mutations are
1646        // visible to the current thread.
1647        self.ref_count.load(Ordering::Acquire) == 1
1648    }
1649}

1963unsafe fn shared_v_to_vec(shared: *mut (), ptr: *const u8, len: usize) -> Vec<u8> {
1964    let shared: *mut Shared = shared.cast();
1965
1966    if (*shared).is_unique() {
1967        let shared = &mut *shared;
1968
1969        // Drop shared
1970        let mut vec = core::mem::take(&mut shared.vec);
1971        release_shared(shared);
1972
1973        // Copy back buffer
1974        ptr::copy(ptr, vec.as_mut_ptr(), len);
1975        vec.set_len(len);
1976
1977        vec
1978    } else {
1979        let v = slice::from_raw_parts(ptr, len).to_vec();
1980        release_shared(shared);
1981        v
1982    }
1983}

1985unsafe fn shared_v_to_mut(shared: *mut (), ptr: *const u8, len: usize) -> BytesMut {
1986    let shared: *mut Shared = shared.cast();
1987
1988    if (*shared).is_unique() {
1989        let shared = &mut *shared;
1990
1991        // The capacity is always the original capacity of the buffer
1992        // minus the offset from the start of the buffer
1993        let v = &mut shared.vec;
1994        let v_capacity = v.capacity();
1995        let v_ptr = v.as_mut_ptr();
1996        let offset = ptr.offset_from(v_ptr) as usize;
1997        let cap = v_capacity - offset;
1998
1999        let ptr = vptr(ptr as *mut u8);
2000
2001        BytesMut {
2002            ptr,
2003            len,
2004            cap,
2005            data: shared,
2006        }
2007    } else {
2008        let v = slice::from_raw_parts(ptr, len).to_vec();
2009        release_shared(shared);
2010        BytesMut::from_vec(v)
2011    }
2012}
